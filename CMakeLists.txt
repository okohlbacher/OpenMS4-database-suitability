cmake_minimum_required(VERSION 3.24)
project(OpenMSDatabaseSuitability VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
include(cmake/OpenMS4Dependencies.cmake)
if(BUILD_TESTING)
  openms4_find_core(COMPONENTS TestSupport)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
else()
  openms4_find_core()
endif()

openms4_find_package(OpenMSCLI)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
add_library(tool_backend STATIC
  source/QC/DBSuitability.cpp
)
target_include_directories(tool_backend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(tool_backend PUBLIC cxx_std_23)
target_link_libraries(tool_backend PUBLIC OpenMS::Core)
target_link_libraries(tool_backend PRIVATE Boost::regex)

include(cmake/OpenMS4Tools.cmake)
openms4_load_tools(package_tools)
set(manifest "# name\tcategory\tversion\texecutable\n")
foreach(tool IN LISTS package_tools)
  add_executable(${tool} src/${tool}.cpp)
  target_compile_features(${tool} PRIVATE cxx_std_23)
  target_link_libraries(${tool} PRIVATE tool_backend OpenMS::CLI)
  openms4_install_rpath(${tool})
  install(TARGETS ${tool} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  string(APPEND manifest "${tool}\t${${tool}_CATEGORY}\t${PROJECT_VERSION}\t${CMAKE_INSTALL_BINDIR}/${tool}${CMAKE_EXECUTABLE_SUFFIX}\n")
  if(BUILD_TESTING)
    foreach(kind ini ctd)
      add_test(NAME ${tool}_write_${kind} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckToolMetadata.py $<TARGET_FILE:${tool}>
        ${tool} ${PROJECT_VERSION} "${${tool}_CATEGORY}" ${kind} ${CMAKE_CURRENT_BINARY_DIR}/metadata/${kind}/${tool}.${kind})
      set_tests_properties(${tool}_write_${kind} PROPERTIES ENVIRONMENT "OPENMS_TOOL_PREFIX_PATH=${CMAKE_BINARY_DIR};OPENMS_DISABLE_UPDATE_CHECK=ON")
    endforeach()
  endif()
endforeach()
file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/share/openms4/tools ${CMAKE_BINARY_DIR}/install-manifests)
string(REPLACE "\t${CMAKE_INSTALL_BINDIR}/" "\tbin/" build_manifest "${manifest}")
file(WRITE ${CMAKE_BINARY_DIR}/share/openms4/tools/database-suitability.tools.tsv "${build_manifest}")
file(WRITE ${CMAKE_BINARY_DIR}/install-manifests/database-suitability.tools.tsv "${manifest}")
install(FILES ${CMAKE_BINARY_DIR}/install-manifests/database-suitability.tools.tsv DESTINATION share/openms4/tools)
if(BUILD_TESTING)
  set(CF_OPENMS_TEST_DATA_PATH "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/")
  configure_file(tests/test_config.h.in generated/OpenMS/test_config.h @ONLY)
  file(GLOB class_tests CONFIGURE_DEPENDS tests/source/*_test.cpp)
  foreach(source IN LISTS class_tests)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} ${source} ${OpenMS_TEST_SUPPORT_SOURCE})
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(${name} PRIVATE tool_backend OpenMS::TestFramework)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()

if(BUILD_TESTING)
  add_executable(DBSuitabilityFakeAdapter tests/source/DBSuitabilityFakeAdapter.cpp)
  target_link_libraries(DBSuitabilityFakeAdapter PRIVATE OpenMS::Core)
  target_compile_definitions(DBSuitabilityProcess_test PRIVATE
    DBSUITABILITY_FAKE_ADAPTER="$<TARGET_FILE:DBSuitabilityFakeAdapter>"
    DBSUITABILITY_CORE_RUNTIME_DIR="$<TARGET_FILE_DIR:OpenMS::Core>")
  add_dependencies(DBSuitabilityProcess_test DBSuitabilityFakeAdapter)
endif()
